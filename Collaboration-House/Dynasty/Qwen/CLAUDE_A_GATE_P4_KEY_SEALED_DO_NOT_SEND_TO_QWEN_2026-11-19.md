    cr = np.cross(dC.T[:, None, :], dD.T[None, :, :]).transpose(2, 0, 1)  # c'(t) x d'(u)
    num = np.einsum('kij,kij->ij', cr, R); den = np.linalg.norm(R, axis=0)**3
